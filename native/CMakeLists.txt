cmake_minimum_required(VERSION 3.22)
project(KotlinCadKernel VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCASCADE 8.0.1 EXACT REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms)
find_package(Eigen3 5.0.1 EXACT REQUIRED NO_MODULE)
add_executable(cad-kernel src/main.cpp)
target_include_directories(cad-kernel PRIVATE ${OpenCASCADE_INCLUDE_DIR} third_party)
target_link_directories(cad-kernel PRIVATE ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(cad-kernel PRIVATE TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKBO TKBool TKFillet TKOffset TKShHealing TKMesh Eigen3::Eigen)
set_target_properties(cad-kernel PROPERTIES INSTALL_RPATH "@loader_path/lib" BUILD_WITH_INSTALL_RPATH FALSE)
install(TARGETS cad-kernel RUNTIME DESTINATION .)
